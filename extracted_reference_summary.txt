const recomputeAllCurriculumDates = (curriculum, timetable, schoolYearStart) => {
            if (!curriculum || curriculum.length === 0) return [];
            if (!timetable || timetable.length === 0 || !schoolYearStart) {
                return curriculum.map(c => ({ ...c }));
            }

            const classIds = [...new Set(curriculum.map(c => c.classId))];
            const updated = [];

            classIds.forEach(cId => {
                const regTkb = timetable
                    .filter(t => t.classId === cId && t.subjectType !== 'specialized')
                    .sort((a, b) => a.dayOfWeek !== b.dayOfWeek ? a.dayOfWeek - b.dayOfWeek : a.period - b.period);

                const specTkb = timetable
                    .filter(t => t.classId === cId && t.subjectType === 'specialized')
                    .sort((a, b) => a.dayOfWeek !== b.dayOfWeek ? a.dayOfWeek - b.dayOfWeek : a.period - b.period);

                const classLessons = curriculum.filter(c => c.classId === cId);
                const weeks = [...new Set(classLessons.map(c => parseInt(c.week)))].sort((a, b) => a - b);

                weeks.forEach(w => {
                    // Regular lessons in week w
                    const regLessons = classLessons
                        .filter(c => parseInt(c.week) === w && c.topic !== 'Chuyên đề')
                        .sort((a, b) => parseInt(a.period) - parseInt(b.period));

                    regLessons.forEach((lesson, idx) => {
                        let expDate = lesson.expectedDate || '';
                        if (regTkb.length > 0) {
                            const tkbSlot = regTkb[idx % regTkb.length];
                            expDate = getDateForWeekDay(w, tkbSlot.dayOfWeek, schoolYearStart);
                        }
                        updated.push({ ...lesson, expectedDate: expDate });
                    });

                    // Specialized lessons in week w
                    const specLessons = classLessons
                        .filter(c => parseInt(c.week) === w && c.topic === 'Chuyên đề')
                        .sort((a, b) => parseInt(a.period) - parseInt(b.period));

                    specLessons.forEach((lesson, idx) => {
                        let expDate = lesson.expectedDate || '';
                        if (specTkb.length > 0) {
                            const tkbSlot = specTkb[idx % specTkb.length];
                            expDate = getDateForWeekDay(w, tkbSlot.dayOfWeek, schoolYearStart);
                        }
                        updated.push({ ...lesson, expectedDate: expDate });
                    });
                });
            });

            const processedIds = new Set(updated.map(u => u.id));
            curriculum.forEach(c => {
                if (!processedIds.has(c.id)) {
                    updated.push({ ...c });
                }
            });

            return updated;
        };

const handleBulkPasteCurriculum = (e) => {
                    e.preventDefault();
                    const targetIds = getTargetClassIds(targetScope);
                    if (targetIds.length === 0) {
                        return showToast('Vui lòng chọn phạm vi lớp áp dụng (hoặc tạo lớp trước khi dán)!', 'error');
                    }

                    const lines = bulkData.split('\n').filter(l => l.trim() !== '');
                    if (lines.length === 0) return showToast('Chưa có dữ liệu dán vào!', 'error');

                    const targetClasses = data.classes.filter(c => targetIds.includes(c.id));
                    const targetNames = targetClasses.map(c => c.name).join(', ');
                    const parsedLessons = [];

                    let currentWeek = 1;
                    let lastMainLessonName = '';
                    let lastSpecLessonName = '';

                    lines.forEach(line => {
                        const parts = line.replace(/\r/g, '').split('\t').map(p => p.trim());

                        // Check if parts[0] is a week indicator
                        if (parts[0]) {
                            const weekMatch = parts[0].match(/^(?:tuần\s*)?(\d{1,2})$/i);
                            if (weekMatch) {
                                const w = parseInt(weekMatch[1]);
                                if (w >= 1 && w <= 52) currentWeek = w;
                            }
                        }

                        // Case A: 4 or 5 columns (Tuần, Tiết CT, Tên bài, Tiết CĐ, Tên chuyên đề)
                        if (parts.length >= 3) {
                            const regPeriodStr = parts[1] || '';
                            const regLessonStr = parts[2] || '';
                            const specPeriodStr = parts[3] || '';
                            const specLessonStr = parts[4] || '';

                            if (regLessonStr) lastMainLessonName = regLessonStr;
                            if (specLessonStr) lastSpecLessonName = specLessonStr;

                            const regPeriod = parseInt(regPeriodStr);
                            const specPeriod = parseInt(specPeriodStr);

                            // Regular lesson
                            if (!isNaN(regPeriod) && regPeriod > 0) {
                                targetIds.forEach(cId => {
                                    parsedLessons.push({
                                        id: generateId(),
                                        classId: cId,
                                        week: currentWeek,
                                        period: regPeriod,
                                        lessonName: regLessonStr || lastMainLessonName,
                                        topic: '',
                                        expectedDate: ''
                                    });
                                });
                            }

                            // Specialized lesson
                            if (!isNaN(specPeriod) && specPeriod > 0) {
                                targetIds.forEach(cId => {
                                    parsedLessons.push({
                                        id: generateId(),
                                        classId: cId,
                                        week: currentWeek,
                                        period: specPeriod,
                                        lessonName: specLessonStr || lastSpecLessonName,
                                        topic: 'Chuyên đề',
                                        expectedDate: ''
                                    });
                                });
                            }
                        }
                        // Case B: 2 columns (Tiết, Tên bài)
                        else if (parts.length === 2) {
                            const pNum = parseInt(parts[0]);
                            const lName = parts[1];
                            if (!isNaN(pNum) && lName) {
                                targetIds.forEach(cId => {
                                    parsedLessons.push({
                                        id: generateId(),
                                        classId: cId,
                                        week: currentWeek,
                                        period: pNum,
                                        lessonName: lName,
                                        topic: '',
                                        expectedDate: ''
                                    });
                                });
                            }
                        }
                    });

                    if (parsedLessons.length === 0) {
                        return showToast('Không tìm thấy bài học hợp lệ. Vui lòng copy trực tiếp các hàng từ Excel!', 'error');
                    }

                    // Remove existing curriculum for target classes
                    const remainingCurriculum = data.curriculum.filter(c => !targetIds.includes(c.classId));
                    const combined = [...remainingCurriculum, ...parsedLessons];

                    // Automatically sync with TKB and SchoolYearStart!
                    const syncedCurriculum = recomputeAllCurriculumDates(combined, data.timetable, data.teacher.schoolYearStart);
                    updateData('curriculum', syncedCurriculum);

                    const regCount = parsedLessons.filter(l => l.topic !== 'Chuyên đề').length / targetIds.length;
                    const specCount = parsedLessons.filter(l => l.topic === 'Chuyên đề').length / targetIds.length;

                    showToast(`Đã dán thành công ${regCount + specCount} tiết (${regCount} chính khóa + ${specCount} chuyên đề) cho lớp [${targetNames}]!`);
                    setBulkData('');
                    setIsEditing(false);
                };
